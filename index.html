const ROOT_DRIVE_FOLDER_ID = "";
const MAIN_FOLDER_NAME = "Gelar Karya SMP Bodhi Dharma";
const SHEET_NAME = "Data_Karya";
const SPAM_LOG_SHEET = "Log_Spam_Terhapus";

const LOCKED_FOLDERS = {
  "7": ["Sanna", "Nyana", "Vinnana", "Sasana"],
  "8": ["Kusala", "Manggala", "Saddha", "Samodana"],
  "9": ["Adhika", "Satti", "Adhicitta", "Magga"]
};

function doGet(e) {
  const action = (e && e.parameter && e.parameter.action) ? e.parameter.action : "list";
  if (action === "list") {
    return jsonResponse({
      status: "success",
      posts: getAllPostsFromSheet(),
      spamLogs: getAllSpamLogsFromSheet()
    });
  }
  return jsonResponse({ status: "ok" });
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(25000);
  try {
    const payload = JSON.parse(e.postData.contents);
    const action = payload.action || "upload";

    if (action === "upload") {
      return handleStudentUpload(payload);
    } else if (action === "view") {
      return handleViewAndAntiSpam(payload);
    } else if (action === "delete") {
      return handleDeletePost(payload.id);
    }

    return jsonResponse({ status: "error", message: "Invalid action" });
  } catch (err) {
    return jsonResponse({ status: "error", message: err.toString() });
  } finally {
    lock.releaseLock();
  }
}

function handleStudentUpload(data) {
  const grade = String(data.grade || "7");
  const folderName = String(data.folder || "Sanna");

  if (!LOCKED_FOLDERS[grade] || LOCKED_FOLDERS[grade].indexOf(folderName) === -1) {
    return jsonResponse({ status: "error", message: "Invalid class folder" });
  }

  const targetFolder = getOrCreateClassFolder(grade, folderName);

  let finalVideoUrl = data.videoUrl || "";
  let driveFileId = "";

  if (data.videoBase64 && data.videoName) {
    const videoBlob = Utilities.newBlob(
      Utilities.base64Decode(data.videoBase64),
      data.videoMimeType || "video/mp4",
      "[" + grade + "-" + folderName + "] " + data.videoName
    );
    const videoFile = targetFolder.createFile(videoBlob);
    videoFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    driveFileId = videoFile.getId();
    finalVideoUrl = "https://drive.google.com/file/d/" + driveFileId + "/preview";
  }

  let finalThumbUrl = data.thumbUrl || "";
  if (data.thumbBase64 && data.thumbName) {
    const thumbBlob = Utilities.newBlob(
      Utilities.base64Decode(data.thumbBase64),
      data.thumbMimeType || "image/jpeg",
      "[THUMB-" + grade + "-" + folderName + "] " + data.thumbName
    );
    const thumbFile = targetFolder.createFile(thumbBlob);
    thumbFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    finalThumbUrl = "https://drive.google.com/thumbnail?id=" + thumbFile.getId() + "&sz=w1000";
  }

  const sheet = getOrCreateMainSheet();
  const id = data.id || ("bd-art-" + new Date().getTime());
  const createdAt = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm");

  sheet.appendRow([
    id,
    grade,
    folderName,
    data.title || "Tanpa Judul",
    data.author || "Siswa SMP Bodhi Dharma",
    data.category || "Seni Budaya",
    1,
    createdAt,
    0,
    finalVideoUrl,
    finalThumbUrl,
    data.description || "",
    driveFileId,
    targetFolder.getUrl()
  ]);

  sortSheetByHighestViews(sheet);

  return jsonResponse({
    status: "success",
    post: {
      id: id,
      grade: grade,
      folder: folderName,
      title: data.title,
      author: data.author,
      category: data.category,
      views: 1,
      createdAt: createdAt,
      lastOpenedAt: 0,
      videoUrl: finalVideoUrl,
      thumbUrl: finalThumbUrl,
      description: data.description,
      driveFileId: driveFileId,
      driveFolderUrl: targetFolder.getUrl()
    }
  });
}

function handleViewAndAntiSpam(data) {
  const sheet = getOrCreateMainSheet();
  const rows = sheet.getDataRange().getValues();
  const nowMs = new Date().getTime();

  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === String(data.id)) {
      const lastOpenedAt = Number(rows[i][8]) || 0;
      const diffSeconds = lastOpenedAt > 0 ? (nowMs - lastOpenedAt) / 1000 : 9999;

      if (diffSeconds < 30) {
        const spamSheet = getOrCreateSpamSheet();
        spamSheet.appendRow([
          Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "HH:mm:ss"),
          diffSeconds.toFixed(1),
          rows[i][0],
          rows[i][1],
          rows[i][2],
          rows[i][3],
          rows[i][4],
          rows[i][6]
        ]);

        const fileId = rows[i][12];
        if (fileId && data.spamPolicy === "delete_post") {
          try { DriveApp.getFileById(fileId).setTrashed(true); } catch (e) {}
        }

        if (data.spamPolicy !== "delete_view") {
          sheet.deleteRow(i + 1);
        }

        return jsonResponse({
          status: "spam_deleted",
          elapsedSec: diffSeconds.toFixed(1)
        });
      }

      const newViews = (Number(rows[i][6]) || 0) + 1;
      sheet.getRange(i + 1, 7).setValue(newViews);
      sheet.getRange(i + 1, 9).setValue(nowMs);
      sortSheetByHighestViews(sheet);

      return jsonResponse({ status: "success", views: newViews });
    }
  }
  return jsonResponse({ status: "not_found" });
}

function handleDeletePost(postId) {
  const sheet = getOrCreateMainSheet();
  const rows = sheet.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === String(postId)) {
      sheet.deleteRow(i + 1);
      return jsonResponse({ status: "success" });
    }
  }
  return jsonResponse({ status: "not_found" });
}

function getOrCreateClassFolder(grade, className) {
  let rootFolder;
  if (ROOT_DRIVE_FOLDER_ID && ROOT_DRIVE_FOLDER_ID.trim() !== "") {
    rootFolder = DriveApp.getFolderById(ROOT_DRIVE_FOLDER_ID.trim());
  } else {
    const iter = DriveApp.getFoldersByName(MAIN_FOLDER_NAME);
    rootFolder = iter.hasNext() ? iter.next() : DriveApp.createFolder(MAIN_FOLDER_NAME);
  }

  const gradeFolderName = "Kelas " + grade;
  const gradeIter = rootFolder.getFoldersByName(gradeFolderName);
  const gradeFolder = gradeIter.hasNext() ? gradeIter.next() : rootFolder.createFolder(gradeFolderName);

  const classFolderName = "Kelas " + grade + " - " + className;
  const classIter = gradeFolder.getFoldersByName(classFolderName);
  const classFolder = classIter.hasNext() ? classIter.next() : gradeFolder.createFolder(classFolderName);

  classFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return classFolder;
}

function getOrCreateMainSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow([
      "ID", "Kelas", "Folder", "Judul Karya", "Kreator", "Kategori",
      "Views", "Waktu Upload", "LastOpenedMs", "Video URL", "Thumbnail URL",
      "Deskripsi Blog", "Drive File ID", "Drive Folder URL"
    ]);
    sheet.getRange(1, 1, 1, 14).setFontWeight("bold").setBackground("#bae6fd");
  }
  return sheet;
}

function getOrCreateSpamSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SPAM_LOG_SHEET);
  if (!sheet) {
    sheet = ss.insertSheet(SPAM_LOG_SHEET);
    sheet.appendRow(["Waktu Terdeteksi", "Selisih Detik", "ID", "Kelas", "Folder", "Judul Karya", "Kreator", "Views Terakhir"]);
    sheet.getRange(1, 1, 1, 8).setFontWeight("bold").setBackground("#fde68a");
  }
  return sheet;
}

function sortSheetByHighestViews(sheet) {
  const lastRow = sheet.getLastRow();
  if (lastRow > 2) {
    sheet.getRange(2, 1, lastRow - 1, 14).sort({ column: 7, ascending: false });
  }
}

function getAllPostsFromSheet() {
  const sheet = getOrCreateMainSheet();
  const rows = sheet.getDataRange().getValues();
  const list = [];
  for (let i = 1; i < rows.length; i++) {
    if (!rows[i][0]) continue;
    list.push({
      id: String(rows[i][0]),
      grade: String(rows[i][1]),
      folder: String(rows[i][2]),
      title: String(rows[i][3]),
      author: String(rows[i][4]),
      category: String(rows[i][5]),
      views: Number(rows[i][6]) || 0,
      createdAt: String(rows[i][7]),
      lastOpenedAt: Number(rows[i][8]) || 0,
      videoUrl: String(rows[i][9] || ""),
      thumbUrl: String(rows[i][10] || ""),
      description: String(rows[i][11] || ""),
      driveFileId: String(rows[i][12] || ""),
      driveFolderUrl: String(rows[i][13] || "")
    });
  }
  return list;
}

function getAllSpamLogsFromSheet() {
  const sheet = getOrCreateSpamSheet();
  const rows = sheet.getDataRange().getValues();
  const logs = [];
  for (let i = 1; i < rows.length; i++) {
    if (!rows[i][0]) continue;
    logs.push({
      logId: "spam-" + i,
      detectedAt: String(rows[i][0]),
      elapsedSec: String(rows[i][1]),
      postBackup: {
        id: String(rows[i][2]),
        grade: String(rows[i][3]),
        folder: String(rows[i][4]),
        title: String(rows[i][5]),
        author: String(rows[i][6]),
        views: Number(rows[i][7]) || 1
      }
    });
  }
  return logs;
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
